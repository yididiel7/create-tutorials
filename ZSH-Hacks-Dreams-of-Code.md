### 1. Edit Command Buffer
```bash
# Open current command in $EDITOR (Ctrl+X, then e)
autoload -Uz edit-command-line
zle -N edit-command-line
bindkey '^Xe' edit-command-line  # Ctrl+X, then e

# Vi mode alternative
# bindkey -M vicmd 'v' edit-command-line
```

### 2. Undo in Zsh
```bash
# Ctrl+_ (Ctrl+Underscore) - built-in undo
# Redo (no default binding):
bindkey '^Y' redo
```

### 3. Magic Space - History Expansion
```bash
# Expand !!, !$, etc. on space
# Requires custom widget (zshcontrib not on Arch):
magic-space() {
    zle history-expand-word
    zle self-insert
}
zle -N magic-space
bindkey ' ' magic-space
```

### 4. chpwd Hook - Run on Directory Change
```bash
# Auto-list on cd
chpwd() { ls }

# Auto-activate Python venv
chpwd() {
  [[ -d .venv ]] && source .venv/bin/activate
}

# Auto-load Nix dev shell
chpwd() {
  [[ -f flake.nix && -z $IN_NIX_SHELL ]] && nix develop
}

# Auto-use Node version
chpwd() { [[ -f .nvmrc ]] && nvm use }
```

### 4.1 Merge Multiple chpwd Hooks
```bash
autoload -Uz add-zsh-hook

function auto_venv() {
  [[ -n $VIRTUAL_ENV && $PWD != *${VIRTUAL_ENV:h}* ]] && deactivate && return
  [[ -n $VIRTUAL_ENV ]] && return
  local dir=$PWD
  while [[ $dir != / ]]; do
    [[ -f $dir/.venv/bin/activate ]] && source $dir/.venv/bin/activate && return
    dir=${dir:h}
  done
}

function auto_nix() {
  [[ -n $IN_NIX_SHELL ]] && return
  local dir=$PWD
  while [[ $dir != / ]]; do
    [[ -f $dir/flake.nix ]] && { [[ ! -f $dir/.envrc ]] && cat > $dir/.envrc <<'EOF'
# autogenerated: load flake dev environment
eval "$(nix print-dev-env)"
EOF
      command direnv allow $dir >/dev/null 2>&1
      command direnv reload >/dev/null 2>&1
      return
    }
    dir=${dir:h}
  done
}

function auto_nvm() { [[ -f .nvmrc ]] && nvm use }

add-zsh-hook chpwd auto_venv
add-zsh-hook chpwd auto_nix
add-zsh-hook chpwd auto_nvm
```

### 9. Custom Widgets
```bash
# Clear screen + scrollback (Ctrl+X, l)
function clear-screen-and-scrollback() {
  echoti civis >$TTY
  printf '%b' '\e[H\e[2J\e[3J' >$TTY
  echoti cnorm >$TTY
  zle redisplay
}
zle -N clear-screen-and-scrollback
bindkey '^Xl' clear-screen-and-scrollback

# Copy buffer to clipboard (Ctrl+X, c)
function copy-buffer-to-clipboard() {
  echo -n "$BUFFER" | pbcopy
  zle -M "Copied to clipboard"
}
zle -N copy-buffer-to-clipboard
bindkey '^Xc' copy-buffer-to-clipboard

# Linux: use wl-copy or xclip
```

## The "Blueprint" (The Why/Logic)

### Key Binding Patterns
| Pattern | Use Case |
|---------|----------|
| `bindkey '^Xe'` | Edit buffer (Ctrl+X prefix) |
| `bindkey -s` | Insert text snippet |
| `bindkey -M vicmd` | Vi command mode |
| `add-zsh-hook` | Merge multiple hooks |

## The "Deep Dive" (The Technical Root)

### ZLE (Zsh Line Editor)
```
Keystroke -> bindkey -> Widget -> Action
                ^
             zle -N
```

### Zmv Pattern Matching
```bash
zmv '(*).log' '$1.txt'      # Capture group $1
zmv -w '*.log' '*.txt'      # Wildcard match
zmv -n                      # Dry run
```

### Named Directory Hash
```bash
hash -d name=path
cd ~name      # Expands to path
ls ~name      # Works anywhere
```

### Suffix/Global Aliases
```bash
alias -s ext=cmd    # suffix: filename.ext runs cmd
alias -g NAME=val   # global: expands anywhere in cmd
```
